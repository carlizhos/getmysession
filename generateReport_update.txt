const generateReport = async () => {
  if (!bulletPoints.trim()) {
    toast.error("Por favor ingresa puntos de la sesión o sube una nota.");
    return;
  }

  setIsProcessing(true);
  setGeneratedReport(null);
  toast.info("Generando reporte SOAP con IA...", { duration: 3000 });

  try {
    // Call Groq AI to generate SOAP report
    const { data, error } = await supabase.functions.invoke('process-clinical-note', {
      body: { 
        text: bulletPoints, 
        action: 'generate_soap' 
        }
    });

    if (error) {
      console.error('AI Error:', error);
      throw new Error(error.message || 'Error al generar reporte');
    }

    setGeneratedReport(data.report);
    toast.success("Reporte SOAP generado con IA", { icon: <Sparkles className="h-4 w-4" /> });

  } catch (error: any) {
    console.error('Report Generation Error:', error);
    toast.error("Error al generar reporte: " + error.message);
  } finally {
    setIsProcessing(false);
  }
};
